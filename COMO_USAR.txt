🔧 PROBLEMAS COMUNS:

   ❌ "Porta já em uso":
      ./stop.sh
      ./start.sh

   ❌ "Ícones não aparecem":
      rm -rf client/.vite
      ./start.sh

   ❌ "Erro no banco de dados":
      Faça backup primeiro:
      cp data/oficina.sqlite data/backup.sqlite
      
      Depois delete e reinicie:
      rm data/oficina.sqlite
      ./start.sh
